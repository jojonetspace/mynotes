-- 2. 刷存量：从 data JSON 解析 has_variant 写入新列（仅处理尚未写值的行）
UPDATE ShopItem_1
SET has_variant = CASE WHEN JSON_VALUE(data, '$.has_variant') IN ('true', '1') THEN 1 ELSE 0 END

版本二：自动遍历所有 `ShopItem_%` 分表（推荐，一次刷完全部分表）

sql

```
-- 对所有名为 ShopItem_数字 的分表：先加列（跳过已存在的），再刷存量数据DECLARE @tbl SYSNAME, @sql NVARCHAR(MAX);
DECLARE tbl_cursor CURSOR FOR    SELECT name FROM sys.tables    WHERE name LIKE 'ShopItem[_]%'      AND name NOT LIKE 'ShopItem[_]Sku%'   -- 排除 SKU 分表      AND ISNUMERIC(REPLACE(name, 'ShopItem_', '')) = 1;  -- 只匹配 ShopItem_数字
OPEN tbl_cursor;FETCH NEXT FROM tbl_cursor INTO @tbl;
WHILE @@FETCH_STATUS = 0BEGIN    -- 加列（列不存在时才加）    IF NOT EXISTS (        SELECT 1 FROM sys.columns        WHERE object_id = OBJECT_ID(@tbl) AND name = 'has_variant')    BEGIN        SET @sql = N'ALTER TABLE ' + QUOTENAME(@tbl) + N' ADD has_variant BIT NULL;';        EXEC sp_executesql @sql;    END
    -- 刷存量（仅处理新列为 NULL 的行）    SET @sql = N'UPDATE ' + QUOTENAME(@tbl) +               N' SET has_variant = CASE WHEN JSON_VALUE(data, ''$.has_variant'') IN (''true'', ''1'') THEN 1 ELSE 0 END' +               N' WHERE has_variant IS NULL AND ISJSON(data) = 1;';    EXEC sp_executesql @sql;
    FETCH NEXT FROM tbl_cursor INTO @tbl;END
CLOSE tbl_cursor;DEALLOCATE tbl_cursor;
```
